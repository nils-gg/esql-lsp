===
two block comments with live code between
===

CREATE COMPUTE MODULE CommentSwallow
	CREATE FUNCTION Main() RETURNS BOOLEAN
	BEGIN
		/* first block */
		DECLARE a INTEGER;
		/* second block */
		DECLARE b INTEGER;
		SET b = a + 1;
		RETURN TRUE;
	END;
END MODULE;
---
