CREATE FUNCTION Main() RETURNS BOOLEAN
BEGIN
  L1:
  WHILE x > 0 DO
    IF x < 0 THEN
      ITERATE L1;
    END IF;
    LEAVE L1;
  END WHILE L1;
  RETURN TRUE;
END
